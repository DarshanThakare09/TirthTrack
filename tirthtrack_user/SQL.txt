-- TirthTrack
-- Phase 1.2 - SQL Migration (Part 1)
--
-- Contains:
-- • Extensions
-- • Enums
-- • Core Tables
-- • Indexes
-- • Constraints
--
-- Safe to execute on a fresh Supabase database.

BEGIN;

-- ============================================================
-- EXTENSIONS
-- ============================================================

CREATE EXTENSION IF NOT EXISTS pgcrypto;

-- Optional (recommended for future GIS support)
CREATE EXTENSION IF NOT EXISTS postgis;

-- ============================================================
-- ENUMS
-- ============================================================

CREATE TYPE gender_enum AS ENUM (
    'male',
    'female',
    'other'
);

CREATE TYPE user_role_enum AS ENUM (
    'user',
    'police',
    'admin'
);

CREATE TYPE police_status_enum AS ENUM (
    'pending',
    'verified',
    'rejected'
);

CREATE TYPE location_source_enum AS ENUM (
    'gps',
    'network',
    'manual'
);

-- ============================================================
-- PROFILES
-- ============================================================

CREATE TABLE profiles (

    id UUID PRIMARY KEY
        REFERENCES auth.users(id)
        ON DELETE CASCADE,

    full_name TEXT NOT NULL,

    mobile VARCHAR(15)
        UNIQUE,

    email TEXT
        UNIQUE,

    gender gender_enum,

    date_of_birth DATE,

    profile_photo TEXT,

    address TEXT,

    city TEXT,

    state TEXT,

    country TEXT
        DEFAULT 'India',

    role user_role_enum
        NOT NULL
        DEFAULT 'user',

    is_active BOOLEAN
        NOT NULL
        DEFAULT TRUE,

    created_at TIMESTAMPTZ
        NOT NULL
        DEFAULT NOW(),

    updated_at TIMESTAMPTZ
        NOT NULL
        DEFAULT NOW()

);

-- ============================================================
-- POLICE DETAILS
-- ============================================================

CREATE TABLE police_details (

    id UUID PRIMARY KEY
        DEFAULT gen_random_uuid(),

    profile_id UUID
        NOT NULL
        UNIQUE
        REFERENCES profiles(id)
        ON DELETE CASCADE,

    badge_number VARCHAR(50)
        NOT NULL
        UNIQUE,

    police_station TEXT
        NOT NULL,

    district TEXT
        NOT NULL,

    state TEXT
        NOT NULL,

    designation TEXT,

    department TEXT,

    id_card_photo TEXT,

    verification_status police_status_enum
        NOT NULL
        DEFAULT 'pending',

    verified_by UUID,

    verified_at TIMESTAMPTZ,

    remarks TEXT,

    created_at TIMESTAMPTZ
        NOT NULL
        DEFAULT NOW(),

    updated_at TIMESTAMPTZ
        NOT NULL
        DEFAULT NOW()

);

-- ============================================================
-- ADMIN DETAILS
-- ============================================================

CREATE TABLE admin_details (

    id UUID PRIMARY KEY
        DEFAULT gen_random_uuid(),

    profile_id UUID
        NOT NULL
        UNIQUE
        REFERENCES profiles(id)
        ON DELETE CASCADE,

    employee_code VARCHAR(50)
        UNIQUE,

    designation TEXT,

    permissions JSONB
        DEFAULT '{}'::jsonb,

    created_at TIMESTAMPTZ
        NOT NULL
        DEFAULT NOW(),

    updated_at TIMESTAMPTZ
        NOT NULL
        DEFAULT NOW()

);

-- ============================================================
-- LIVE LOCATIONS
-- ============================================================

CREATE TABLE live_locations (

    id UUID PRIMARY KEY
        DEFAULT gen_random_uuid(),

    profile_id UUID
        NOT NULL
        REFERENCES profiles(id)
        ON DELETE CASCADE,

    latitude DOUBLE PRECISION
        NOT NULL,

    longitude DOUBLE PRECISION
        NOT NULL,

    accuracy DOUBLE PRECISION,

    altitude DOUBLE PRECISION,

    speed DOUBLE PRECISION,

    heading DOUBLE PRECISION,

    battery_percentage SMALLINT,

    location_source location_source_enum
        DEFAULT 'gps',

    recorded_at TIMESTAMPTZ
        NOT NULL
        DEFAULT NOW(),

    CONSTRAINT battery_percentage_check
        CHECK (
            battery_percentage IS NULL
            OR
            (battery_percentage BETWEEN 0 AND 100)
        )

);

-- ============================================================
-- INDEXES
-- ============================================================

CREATE INDEX idx_profiles_mobile
ON profiles(mobile);

CREATE INDEX idx_profiles_email
ON profiles(email);

CREATE INDEX idx_profiles_role
ON profiles(role);

CREATE INDEX idx_police_profile
ON police_details(profile_id);

CREATE INDEX idx_police_station
ON police_details(police_station);

CREATE INDEX idx_police_status
ON police_details(verification_status);

CREATE INDEX idx_admin_profile
ON admin_details(profile_id);

CREATE INDEX idx_live_locations_profile
ON live_locations(profile_id);

CREATE INDEX idx_live_locations_time
ON live_locations(recorded_at DESC);

CREATE INDEX idx_live_locations_profile_time
ON live_locations(profile_id, recorded_at DESC);

CREATE INDEX idx_live_locations_coordinates
ON live_locations(latitude, longitude);

-- ============================================================
-- FOREIGN KEY
-- (verified_by references admin_details)
-- Added after table creation to avoid dependency issues.
-- ============================================================

ALTER TABLE police_details
ADD CONSTRAINT fk_police_verified_by
FOREIGN KEY (verified_by)
REFERENCES admin_details(id)
ON DELETE SET NULL;

-- ============================================================
-- COMMENTS
-- ============================================================

COMMENT ON TABLE profiles IS
'Stores user profile information.';

COMMENT ON TABLE police_details IS
'Stores police verification details.';

COMMENT ON TABLE admin_details IS
'Stores administrator information.';

COMMENT ON TABLE live_locations IS
'Stores real-time GPS locations.';

-- ============================================================
-- ROUTES
-- ============================================================

CREATE TABLE routes (

    id UUID PRIMARY KEY
        DEFAULT gen_random_uuid(),

    route_name TEXT NOT NULL,

    route_code VARCHAR(50)
        UNIQUE,

    description TEXT,

    total_distance_km NUMERIC(8,2),

    estimated_time_minutes INTEGER,

    is_active BOOLEAN
        DEFAULT TRUE,

    created_at TIMESTAMPTZ
        DEFAULT NOW(),

    updated_at TIMESTAMPTZ
        DEFAULT NOW()

);

-- ============================================================
-- ROUTE NODES
-- ============================================================

CREATE TABLE route_nodes (

    id UUID PRIMARY KEY
        DEFAULT gen_random_uuid(),

    route_id UUID NOT NULL
        REFERENCES routes(id)
        ON DELETE CASCADE,

    node_order INTEGER NOT NULL,

    node_name TEXT NOT NULL,

    latitude DOUBLE PRECISION NOT NULL,

    longitude DOUBLE PRECISION NOT NULL,

    created_at TIMESTAMPTZ
        DEFAULT NOW(),

    CONSTRAINT uq_route_node_order
        UNIQUE(route_id,node_order)

);

    updated_at TIMESTAMPTZ
        DEFAULT NOW()

);

        DEFAULT TRUE,

    created_at TIMESTAMPTZ
        DEFAULT NOW(),

    updated_at TIMESTAMPTZ
        DEFAULT NOW()

);

    id UUID PRIMARY KEY
        DEFAULT gen_random_uuid(),

    title TEXT NOT NULL,

    message TEXT NOT NULL,

    alert_type alert_type_enum
        NOT NULL
        DEFAULT 'general',

    priority alert_priority_enum
        NOT NULL
        DEFAULT 'medium',

    created_by UUID
        REFERENCES admin_details(id)
        ON DELETE SET NULL,

    expires_at TIMESTAMPTZ,

    is_active BOOLEAN
        NOT NULL
        DEFAULT TRUE,

-- ============================================================
-- ALERT READS
-- ============================================================

CREATE TABLE alert_reads (

    id UUID PRIMARY KEY
        DEFAULT gen_random_uuid(),

    alert_id UUID NOT NULL
        REFERENCES alerts(id)
        ON DELETE CASCADE,

    profile_id UUID NOT NULL
        REFERENCES profiles(id)
        ON DELETE CASCADE,

    read_at TIMESTAMPTZ
        NOT NULL
        DEFAULT NOW(),

    CONSTRAINT uq_alert_read
        UNIQUE(alert_id, profile_id)

);

-- ============================================================
-- NOTIFICATIONS
-- ============================================================

CREATE TABLE notifications (

    id UUID PRIMARY KEY
        DEFAULT gen_random_uuid(),

    profile_id UUID

    alert_id UUID
        REFERENCES alerts(id)
        ON DELETE SET NULL,

    title TEXT NOT NULL,

    body TEXT NOT NULL,

    status notification_status_enum
        NOT NULL
        DEFAULT 'pending',

    is_read BOOLEAN
        NOT NULL
        DEFAULT FALSE,

    sent_at TIMESTAMPTZ,

    read_at TIMESTAMPTZ,

    created_at TIMESTAMPTZ
        NOT NULL
        DEFAULT NOW()

);

        NOT NULL
        DEFAULT NOW(),

    updated_at TIMESTAMPTZ

        DEFAULT NOW(),

CREATE TABLE chatbot_sessions (

    id UUID PRIMARY KEY
        DEFAULT gen_random_uuid(),

    profile_id UUID
        NOT NULL
        REFERENCES profiles(id)
        ON DELETE CASCADE,

    session_title TEXT,

    status chat_session_status_enum
        NOT NULL
        DEFAULT 'active',

    started_at TIMESTAMPTZ
        NOT NULL
        DEFAULT NOW(),

    ended_at TIMESTAMPTZ,

    created_at TIMESTAMPTZ
        NOT NULL
        DEFAULT NOW(),

    updated_at TIMESTAMPTZ
        NOT NULL
        DEFAULT NOW()

);

-- ============================================================
-- CHATBOT MESSAGES
-- ============================================================

CREATE TABLE chatbot_messages (

    id UUID PRIMARY KEY
        DEFAULT gen_random_uuid(),

    session_id UUID
        NOT NULL
        REFERENCES chatbot_sessions(id)
        ON DELETE CASCADE,

    sender chat_message_role_enum
        NOT NULL,

    message TEXT
        NOT NULL,

    metadata JSONB
        DEFAULT '{}'::jsonb,

    created_at TIMESTAMPTZ
        NOT NULL
        DEFAULT NOW()

);

-- ============================================================
-- APP SETTINGS
-- ============================================================

CREATE TABLE app_settings (

    id UUID PRIMARY KEY
        DEFAULT gen_random_uuid(),

    setting_key VARCHAR(150)
        NOT NULL
        UNIQUE,

    setting_value JSONB
        NOT NULL,

    description TEXT,

    is_public BOOLEAN
        NOT NULL
        DEFAULT FALSE,

    created_by UUID
        REFERENCES admin_details(id)
        ON DELETE SET NULL,

    created_at TIMESTAMPTZ
        NOT NULL
        DEFAULT NOW(),

    updated_at TIMESTAMPTZ
        NOT NULL
        DEFAULT NOW()

);

-- ============================================================
-- INDEXES
-- ============================================================

CREATE INDEX idx_chatbot_sessions_profile
ON chatbot_sessions(profile_id);

CREATE INDEX idx_chatbot_sessions_status
ON chatbot_sessions(status);

CREATE INDEX idx_chatbot_sessions_started
ON chatbot_sessions(started_at DESC);

CREATE INDEX idx_chatbot_messages_session
ON chatbot_messages(session_id);

CREATE INDEX idx_chatbot_messages_sender
ON chatbot_messages(sender);

CREATE INDEX idx_chatbot_messages_created
ON chatbot_messages(created_at);

CREATE INDEX idx_app_settings_public
ON app_settings(is_public);

CREATE INDEX idx_app_settings_key
ON app_settings(setting_key);

-- Current user's application role
CREATE OR REPLACE FUNCTION public.current_user_role()
RETURNS user_role_enum
LANGUAGE sql
STABLE
SECURITY DEFINER
SET search_path = public
AS $$
    SELECT role
    FROM profiles
    WHERE id = auth.uid();
$$;

-- Is current user Admin?
CREATE OR REPLACE FUNCTION public.is_admin()
RETURNS BOOLEAN
LANGUAGE sql
STABLE
SECURITY DEFINER
SET search_path = public
AS $$
    SELECT EXISTS (
        SELECT 1
        FROM admin_details ad
        JOIN profiles p
            ON p.id = ad.profile_id
        WHERE p.id = auth.uid()
    );
$$;

-- Is current user Police?
CREATE OR REPLACE FUNCTION public.is_police()
RETURNS BOOLEAN
LANGUAGE sql
STABLE
SECURITY DEFINER
SET search_path = public
AS $$
    SELECT EXISTS (
        SELECT 1
        FROM police_details pd
        JOIN profiles p
            ON p.id = pd.profile_id
        WHERE p.id = auth.uid()
          AND pd.verification_status = 'verified'
    );
$$;

CREATE OR REPLACE FUNCTION public.handle_new_user()
RETURNS trigger
LANGUAGE plpgsql
SECURITY DEFINER
SET search_path = public
AS $$
BEGIN

    INSERT INTO public.profiles (
        id,
        mobile,
        role,
        is_active,
        created_at,
        updated_at
    )
    VALUES (
        NEW.id,
        NEW.phone,
        'user',
        TRUE,
        NOW(),
        NOW()
    )
    ON CONFLICT (id) DO NOTHING;

    RETURN NEW;

END;
$$;

AFTER INSERT
ON auth.users
FOR EACH ROW
EXECUTE FUNCTION public.handle_new_user();
